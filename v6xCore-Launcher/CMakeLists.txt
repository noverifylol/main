cmake_minimum_required(VERSION 3.20)
project(v6xCore VERSION 3.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 99)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_library(monocypher STATIC vendor/monocypher/monocypher.c vendor/monocypher/monocypher-ed25519.c)
target_include_directories(monocypher PUBLIC vendor/monocypher)
foreach(name math profile_name data team_rules license_protocol)
    add_executable(test_${name} tests/test_${name}.cpp)
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
    if(MSVC)
        target_compile_options(test_${name} PRIVATE /utf-8)
    endif()
endforeach()
target_link_libraries(test_license_protocol PRIVATE monocypher)
if(WIN32)
    enable_language(RC)
    find_package(Threads REQUIRED)
    add_library(imgui STATIC vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp
        vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp
        vendor/imgui/backends/imgui_impl_dx11.cpp vendor/imgui/backends/imgui_impl_win32.cpp)
    target_include_directories(imgui PUBLIC vendor/imgui vendor/imgui/backends)
    target_compile_definitions(imgui PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
    add_executable(v6xCore WIN32 src/main.cpp assets/app.rc)
    set_target_properties(v6xCore PROPERTIES OUTPUT_NAME "v6xCore Launcher")
    target_include_directories(v6xCore PRIVATE assets ${CMAKE_SOURCE_DIR})
    target_compile_definitions(v6xCore PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(v6xCore PRIVATE imgui monocypher Threads::Threads d3d11 d3dcompiler dxgi dwmapi
        gdiplus ole32 uuid shell32 user32 gdi32 imm32 winhttp bcrypt crypt32 advapi32)
    foreach(name settings bootstrap license_cache target_selection)
        add_executable(test_${name} tests/test_${name}.cpp)
        target_compile_definitions(test_${name} PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
        target_link_libraries(test_${name} PRIVATE monocypher Threads::Threads user32 shell32 ole32 uuid winhttp bcrypt crypt32 advapi32)
        add_test(NAME ${name} COMMAND test_${name})
        if(MSVC)
            target_compile_options(test_${name} PRIVATE /utf-8)
        else()
            target_link_options(test_${name} PRIVATE -static -static-libgcc -static-libstdc++)
        endif()
    endforeach()
    target_sources(test_bootstrap PRIVATE assets/app.rc)
    target_include_directories(test_bootstrap PRIVATE assets ${CMAKE_SOURCE_DIR})
    if(MSVC)
        target_compile_options(v6xCore PRIVATE /W4 /permissive- /utf-8 /guard:cf)
        target_link_options(v6xCore PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
    else()
        target_compile_options(v6xCore PRIVATE -Wall -Wextra -Wpedantic)
        target_link_options(v6xCore PRIVATE -municode -static -static-libgcc -static-libstdc++ -s
            -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
    endif()
    # Runtime defaults and artwork are embedded. Optional developer data is copied for offline rebuilds.
    add_custom_command(TARGET v6xCore POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/reference" "$<TARGET_FILE_DIR:v6xCore>/reference"
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:v6xCore>/tools"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/tools/update_offsets.py" "$<TARGET_FILE_DIR:v6xCore>/tools/update_offsets.py")
endif()
